** start of script.js **

const fIcon = document.querySelectorAll(".favorite-icon");

function filled(i) {
  if (i.classList.contains("filled") === false) {
    i.classList.add("filled");
    i.innerHTML = `&#10084;`;
  } else {
    i.classList.remove("filled");
    i.innerHTML = `&#9825;`;
  }
}

fIcon.forEach(i => {
  i.addEventListener("click", () => filled(i));
})
