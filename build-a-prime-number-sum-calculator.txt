function sumPrimes(num) {
  let sum = 0;
  if (num < 2) {
    return sum;
  } else {
    let numArr = new Array;
    for (let i = 0; i <= num; i++){
      numArr = [];
      for (let j = 2; j < i; j++) {
        numArr.push(j)
      }
      if (numArr.every(key => {
        if (i % key !== 0) {
          return true;
        }
      })) {
        console.log(i);
        sum += i;
      }
    }
  }
  return sum-1;
}
