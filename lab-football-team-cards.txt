** start of script.js **

let footballTeam = {
  team: "Olympiakos",
  year: 2000,
  headCoach: "Coach Coch",
  players: [
    {
      name: "name1",
      position: "forward",
      isCaptain: true
    },
    {
      name: "name2",
      position: "midfielder",
      isCaptain: false
    },
    {
      name: "name3",
      position: "defender",
      isCaptain: false
    },
    {
      name: "name4",
      position: "goalkeeper",
      isCaptain: false
    },
  ]
}

let headCoach = document.getElementById("head-coach");
let team = document.getElementById("team");
let year = document.getElementById("year");
let playerCards = document.getElementById("player-cards");

headCoach.textContent = footballTeam.headCoach;
team.textContent = footballTeam.team;
year.textContent = footballTeam.year;



function getPlayerCards(position) {

  footballTeam.players.map((i) => {
    if (position === "all") { 
      let playerCardDiv = document.createElement("div");
      let playerName = document.createElement("h2");
      let playerPosition = document.createElement("p");

      playerCardDiv.setAttribute("class", "player-card");
      playerCardDiv.appendChild(playerName);
      playerCardDiv.appendChild(playerPosition);

      playerCards.appendChild(playerCardDiv);
      playerPosition.textContent = `Position: ${i.position}`;

      if (i.isCaptain) {
        playerName.textContent = `(Captain) ${i.name}`
      } else {
        playerName.textContent = `${i.name}`
      }
    } else {
      let playerCardDiv = document.createElement("div");
      let playerName = document.createElement("h2");
      let playerPosition = document.createElement("p");

      playerCardDiv.setAttribute("class", "player-card");
      if (i.position === position){
        playerCardDiv.appendChild(playerName);
        playerCardDiv.appendChild(playerPosition);

        playerCards.appendChild(playerCardDiv);
        playerPosition.textContent = `Position: ${i.position}`;

        if (i.isCaptain) {
          playerName.textContent = `(Captain) ${i.name}`
        } else {
          playerName.textContent = `${i.name}`
        }
      }
      }
  });

}

function resetCardsDiv() {
  playerCards.innerHTML = "";
}

let playerSelect = document.getElementById("players");
let playerCard = document.querySelectorAll(".player-card");

getPlayerCards("all");
playerSelect.addEventListener("change", () => {
  switch (event.target.value) {
    case "all":
      resetCardsDiv();
      getPlayerCards("all");
      break;
    case "forward":
      resetCardsDiv();
      getPlayerCards("forward");
      break;
    case "midfielder":
      resetCardsDiv();
      getPlayerCards("midfielder");
      break;
    case "defender":
      resetCardsDiv();
      getPlayerCards("defender");
      break;
    case "goalkeeper":
      resetCardsDiv();
      getPlayerCards("goalkeeper");
      break;
    default:
      resetCardsDiv();
      getPlayerCards("all")
  }
})
