function steamrollArray(arr) {

  let flatArr = new Array;
  
  for(let i of arr){
    if(i === ""){
      arr.splice(i,1);
    } else if (typeof i === "number") {
      flatArr.push(i);
    } else if (typeof i === "string") {
      flatArr.push(i);
    } else if (typeof i === "object" && i !== null && Array.isArray(i) === false) {
      flatArr.push(i);
    } else if (Array.isArray(i)) {
      for (let j of i) {
        if (j === ""){
            arr.splice(j,1);
          } else if (typeof j === "number") {
            flatArr.push(j);
          } else if (typeof j === "string") {
            flatArr.push(j);
          } else if (typeof j === "object" && j !== null && Array.isArray(j) === false) {
            flatArr.push(j);
          } else if (Array.isArray(j)) {
            for (let k of j) {
              if (k === ""){
                  arr.splice(k,1);
                } else if (typeof k === "number") {
                  flatArr.push(k);
                } else if (typeof k === "string") {
                  flatArr.push(k);
                } else if (typeof k === "object" && k !== null && Array.isArray(k) === false) {
                  flatArr.push(k);
                } else if (Array.isArray(k)) {
                  flatArr.push(k[0])
                }
          }
      }
    }
    }
  }
  return flatArr;
}

console.log(steamrollArray([1, {}, [3, [[4]]]]));
