** start of script.js **

  const txtInput = document.getElementById("text-input");
  const charCount = document.getElementById("char-count");

function txtAreaLength(input) {
  return input.value.length;
}

function txtCount() {
  let txtLength = txtAreaLength(txtInput);
  if (txtLength >= 50) {
    charCount.style.color = "red";
    txtInput.value = txtInput.value.slice(0,50);
  } else {
    charCount.style.color = "black";

  }

  charCount.textContent = `Character Count: ${txtLength}/50`;
}

txtInput.addEventListener("input", txtCount)
